void reverse_array(int arr[], int size) {
  int start = 0;
  int end = size - 1;
  int temp = 0;

  while (start < end) {
    temp = arr[start];
    arr[start] = arr[end];
    arr[end] = temp;

    start++;
    end--;
  }
}
