int sumTail(int arr[], int startIdx, int endIdx, int accumulator) {
  if (startIdx >= endIdx) {
    return accumulator;
  }
  return sumTail(arr, startIdx + 1, endIdx, accumulator + arr[startIdx]);
}

int arraySum(int arr[], int n) {

    return 0;
  }

}

}
